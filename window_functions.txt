ROW_NUMBER()
SELECT 
    employee,
    salary,
    ROW_NUMBER() OVER (ORDER BY salary DESC) AS row_num
FROM employees;


| employee | salary | row_num |
| -------- | ------ | ------- |
| A        | 100k   | 1       |
| B        | 90k    | 2       |
| C        | 90k    | 3       |
| D        | 80k    | 4       |


RANK()
SELECT 
    employee,
    salary,
    RANK() OVER (ORDER BY salary DESC) AS rank
FROM employees;

| employee | salary | rank |
| -------- | ------ | ---- |
| A        | 100k   | 1    |
| B        | 90k    | 2    |
| C        | 80k    | 3    |
| D        | 80k    | 3    |
| E        | 60k    | 5    |


***THE DIFFERENCE BETWEEN ROW_NUMBER() and RANK() IS THAT ROW_NUMBER ASSIGNS A **UNIQUE**
SEQUENTIAL NUMBER TO EACH ROW WHILE RANK() WILL GIVE ROWS WITH THE SAME VALUE THE SAME
RANK
*** NOTE THAT THERE IS A GAP BETWEEN D and E 

DENSE_RANK()
SELECT

Percentile (PERCENT_RANK)
SELECT
    loan_amnt,
    PERCENT_RANK() OVER (ORDER BY loan_amnt) AS pct
FROM loans;

| loan_amnt | rank position | pct  |
| --------- | ------------- | ---- |
| 4,000     | lowest        | 0.00 |
| 5,000     | 2nd           | 0.33 |
| 6,000     | 3rd           | 0.66 |
| 8,000     | highest       | 1.00 |

*** for PERCENT_RANK the ORDER BY value will rank the values using suqential numbers 
and PERCENT_RANK will do the calculation of (rank - 1) / (total_rows - 1)

LAG() (previous row value)
SELECT
    loan_id,
    loan_amnt,
    LAG(loan_amnt) OVER (ORDER BY loan_id) AS previous_loan
FROM loans;

original:                                   
| loan_id | loan_amnt |                     
| ------- | --------- |

| 102     | 6,000     |
| 103     | 8,000     |
| 104     | 4,000     |

after:

SUM() OVER
Running Total (ORDER BY inside window)

AVG() OVER
Moving Averages
SELECT 
    date,
    stock_price,
    AVG(stock_price) OVER (ORDER BY date ROWS BETWEEN 6 PRECEDING AND CURRENT ROW) as seven_day_ma
FROM stock_prices;

NTILE()
Divides dataset into N-equal sized buckets 
PARTITION BY (grouped window)
SELECT
    region,
    customer_id,
    person_income,
    SUM(person_income) OVER (PARTITION BY region) AS region_total_income
FROM customers;

| region | customer_id | person_income | region_total_income |
| ------ | ----------- | ------------- | ------------------- |
| West   | 1           | 50,000        | 130,000             |
| West   | 2           | 80,000        | 130,000             |
| East   | 3           | 60,000        | 100,000             |
| East   | 4           | 40,000        | 100,000             |

*** PARTITION BY IS LIKE GROUP BY BUT IT STILL INCLUDES ALL THE ROWS

FIRST_VALUE()
Returns first value in ordered partition
SELECT 
    product_name,
    date,
    price,
    FIRST_VALUE(price) OVER (PARTITION BY product_name ORDER BY date) as launch_price,
FROM product_prices
ORDER BY date;

CUME_DIST()
returns the cumulative distribution (percentage of values less than or equal to current)
SELECT 
    product_name,
    units_sold,
    CUME_DIST() OVER (ORDER BY units_sold) as cumulative_distribution
FROM products
ORDER BY units_sold;

product_name | units_sold | cumulative_distribution
Product A    | 100        | 0.20  -- 20% of products sold <= 100
Product B    | 150        | 0.40
Product C    | 200        | 0.60
Product D    | 250        | 0.80
Product E    | 300        | 1.00

NTH_VALUE()
Returns the Nth value in an ordered partition.
SELECT 
    date,
    revenue,
    NTH_VALUE(revenue, 2) OVER (
        PARTITION BY EXTRACT(YEAR FROM date) 
        ORDER BY date
        ROWS BETWEEN UNBOUNDED PRECEDING AND UNBOUNDED FOLLOWING
    ) as second_month_revenue
FROM monthly_revenue;

date    | revenue | second_month_revenue
--------|---------|---------------------
2024-01 | 10000   | 12000  -- Feb revenue
2024-02 | 12000   | 12000
2024-03 | 11000   | 12000
2024-04 | 13000   | 12000

Actual Example:
SELECT
    region,
    customer_id,
    person_income,
    RANK() OVER (PARTITION BY region ORDER BY person_income DESC) AS region_income_rank
FROM customers;

| region | customer_id | income | region_income_rank |
| ------ | ----------- | ------ | ------------------ |
| West   | 2           | 80,000 | 1                  |
| West   | 1           | 50,000 | 2                  |
| East   | 3           | 60,000 | 1                  |
| East   | 4           | 40,000 | 2                  |

*** Here we are PARTITIONing BY region so for each region the customers and their income
have separate ranks - i am assuming that if the order was ASC then the rank 1 woukd start 
at the lowest income value
